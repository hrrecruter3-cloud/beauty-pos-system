   119→   119→
   120→   120→  const filtered = useMemo(() => products, [products])
   121→   121→
   122→   122→  const openAdd = () => {

   127→   127→

   135→   135→      categoryId: p.categoryId || '',

   137→   137→      unitId: p.unitId || '',
   138→   138→      supplierId: p.supplierId || '',
   139→   139→      purchaseCost: String(p.purchaseCost ?? 0),
   140→   140→      sellingPrice: String(p.sellingPrice ?? 0),
   141→   141→      wholesalePrice: String(p.wholesalePrice ?? 0),
   142→   142→      taxRate: String(p.taxRate ?? 0),
   143→   143→      minStock: String(p.minStock ?? 0),
   144→   144→      reorderLevel: String(p.reorderLevel ?? 0),
   145→   145→      openingStock: '0',
   146→   146→      image: p.image || '',
   147→   147→      description: p.description || '',
   148→   148→      active: p.active ?? true,
   149→   149→    })
   150→   150→    setDialogOpen(true)
   151→   151→  }
   152→   152→
   153→   153→  const handleSave = async () => {
   154→   154→    if (!form.name || !form.sku) {
   155→   155→      toast.error('الاسم ورمز SKU مطلوبان')
   156→   156→      return
   157→   157→    }
   158→   158→    setSaving(true)
   159→   159→    try {

   168→   168→          method: 'PUT',

   172→   172→      } else {
   173→   173→        await apiFetch('/products', {
   174→   174→          method: 'POST',
   175→   175→          body: JSON.stringify(body),
   176→   176→        })
   177→   177→        toast.success('تم إنشاء المنتج بنجاح')
   178→   178→      }
   179→   179→      setDialogOpen(false)
   180→   180→      loadProducts()
   181→   181→    } catch (e: any) {
   182→   182→      toast.error(e.message)
   183→   183→    } finally {
   184→   184→      setSaving(false)
   185→   185→    }
   186→   186→  }
   187→   187→
   188→   188→  const handleArchive = async (p: any) => {
   189→   189→    if (!confirm(`هل أنت متأكد من أرشفة المنتج "${p.nameAr || p.name}"؟`)) return
   190→   190→    try {
   191→   191→      await apiFetch(`/products/${p.id}`, { method: 'DELETE' })
   192→   192→      toast.success('تم أرشفة المنتج')

   202→   202→    const lines = [headers.join(',')]
   203→   203→    rows.forEach((p) => {
   204→   204→      const stock = p.currentStock ?? 0
   205→   205→      const cat = p.category ? (p.category.nameAr || p.category.name) : ''
   206→   206→      const line = [
   207→   207→        `"${p.name || ''}"`,
   208→   208→        `"${p.nameAr || ''}"`,
   209→   209→        `"${p.sku || ''}"`,
   210→   210→        `"${p.barcode || ''}"`,

   212→   212→        p.purchaseCost ?? 0,
   213→   213→        p.sellingPrice ?? 0,
   214→   214→        p.wholesalePrice ?? 0,
   215→   215→        p.taxRate ?? 0,
   216→   216→        stock,
   217→   217→        p.minStock ?? 0,
   218→   218→        p.reorderLevel ?? 0,

   234→   234→  const handleImportClick = () => fileInputRef.current?.click()

   236→   236→  const handleImportFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
   237→   237→    const file = e.target.files?.[0]
   238→   238→    if (!file) return
   239→   239→    try {
   240→   240→      const text = await file.text()
   241→   241→      const lines = text.split(/\r?\n/).filter(Boolean)
   242→   242→      if (lines.length < 2) {
   243→   243→        toast.error('الملف فارغ أو غير صالح')
   244→   244→        return
   245→   245→      }
   246→   246→      // Skip header
   247→   247→      const parsed = lines.slice(1).map((line) => {
   248→   248→        // simple CSV parser handling quoted strings
   249→   249→        const cells: string[] = []
   250→   250→        let cur = ''
   251→   251→        let inQ = false
   252→   252→        for (let i = 0; i < line.length; i++) {
   253→   253→          const ch = line[i]
   254→   254→          if (ch === '"') inQ = !inQ
   255→   255→          else if (ch === ',' && !inQ) { cells.push(cur); cur = '' }
   256→   256→          else cur += ch
   257→   257→        }
   258→   258→        cells.push(cur)
   259→   259→        return cells
   260→   260→      }).map((c) => ({
   261→   261→        name: c[0] || '',
   262→   262→        nameAr: c[1] || '',
   263→   263→        sku: c[2] || '',
   264→   264→        barcode: c[3] || '',
   265→   265→        category: c[4] || '',
   266→   266→        purchaseCost: c[5] || '0',
   267→   267→        sellingPrice: c[6] || '0',
   268→   268→        wholesalePrice: c[7] || '0',
   269→   269→        taxRate: c[8] || '0',
   270→   270→        stock: c[9] || '0',
   271→   271→      }))
   272→   272→      setImportPreview(parsed)
   273→   273→      toast.success(`تم تحميل ${parsed.length} سجل`)
   274→   274→    } catch (err: any) {
   275→   275→      toast.error('فشل قراءة الملف: ' + err.message)
   276→   276→    } finally {
   277→   277→      // reset input so the same file can be re-selected
   278→   278→      if (fileInputRef.current) fileInputRef.current.value = ''
   279→   279→    }
   280→   280→  }
   281→   281→

   288→   288→        await apiFetch('/products', {
   289→   289→          method: 'POST',
   290→   290→          body: JSON.stringify({
   291→   291→            name: row.name,
   292→   292→            nameAr: row.nameAr,
   293→   293→            sku: row.sku || `SKU-${Date.now()}-${ok}`,
   294→   294→            barcode: row.barcode || undefined,
   295→   295→            purchaseCost: row.purchaseCost,
   296→   296→            sellingPrice: row.sellingPrice,
   297→   297→            wholesalePrice: row.wholesalePrice,
   298→   298→            taxRate: row.taxRate,
   299→   299→            openingStock: row.stock,
   300→   300→            active: true,
   301→   301→          }),
   302→   302→        })
   303→   303→        ok++
   304→   304→      } catch {
   305→   305→        fail++
   306→   306→      }
   307→   307→    }
   308→   308→    toast.success(`تم استيراد ${ok} منتج${fail ? `, فشل ${fail}` : ''}`)
   309→   309→    setImportPreview(null)
   310→   310→    loadProducts()
   311→   311→  }
   312→   312→

   319→   319→      return s <= (p.reorderLevel ?? 0) && s > 0
   320→   320→    }).length
   321→   321→    const outOfStock = filtered.filter((p) => (p.currentStock ?? 0) <= 0).length
   322→   322→    return { total, active, lowStock, outOfStock }
   323→   323→  }, [filtered])
   324→   324→
   325→   325→  return (
   326→   326→    <div className="p-4 md:p-6 space-y-6 max-w-[1600px] mx-auto">
   327→   327→      {/* Header */}
   328→   328→      <div className="flex flex-col md:flex-row md:items-center justify-between gap-3">
   329→   329→        <div>
   330→   330→          <h1 className="text-2xl font-bold flex items-center gap-2">
   331→   331→            <Package className="w-6 h-6 text-primary" />
   332→   332→            المنتجات
   333→   333→          </h1>
   334→   334→          <p className="text-muted-foreground text-sm">إدارة المنتجات والأسعار والمخزون</p>

   336→   336→        <div className="flex flex-wrap gap-2">

   338→   338→            ref={fileInputRef}
   339→   339→            type="file"
   340→   340→            accept=".csv"
   341→   341→            className="hidden"
   342→   342→            onChange={handleImportFile}
   343→   343→          />

   353→   353→            <Plus className="w-4 h-4" />
   354→   354→            إضافة منتج

   371→   371→          </CardContent>
   372→   372→        </Card>
   373→   373→        <Card>
   374→   374→          <CardContent className="p-4">
   375→   375→            <p className="text-xs text-muted-foreground">مخزون منخفض</p>

   380→   380→          <CardContent className="p-4">
   381→   381→            <p className="text-xs text-muted-foreground">نفد المخزون</p>

   383→   383→          </CardContent>
   384→   384→        </Card>
   385→   385→      </div>
   386→   386→
   387→   387→      {/* Filters */}
   388→   388→      <Card>
   389→   389→        <CardContent className="p-4">
   390→   390→          <div className="flex flex-col md:flex-row gap-3">
   391→   391→            <div className="relative flex-1">
   392→   392→              <Search className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   393→   393→              <Input
   394→   394→                placeholder="بحث بالاسم أو SKU أو الباركود..."

   396→   396→                onChange={(e) => setSearch(e.target.value)}
   397→   397→                className="pr-10"
   398→   398→              />

   401→   401→              <SelectTrigger className="w-full md:w-56">
   402→   402→                <Filter className="w-4 h-4 ml-1 text-muted-foreground" />
   403→   403→                <SelectValue placeholder="كل الفئات" />
   404→   404→              </SelectTrigger>
   405→   405→              <SelectContent>
   406→   406→                <SelectItem value="all">كل الفئات</SelectItem>
   407→   407→                {categories.map((c) => (
   408→   408→                  <SelectItem key={c.id} value={c.id}>
   409→   409→                    {c.nameAr || c.name}
   410→   410→                  </SelectItem>

   417→   417→              </SelectTrigger>
   418→   418→              <SelectContent>
   419→   419→                <SelectItem value="all">الكل</SelectItem>
   420→   420→                <SelectItem value="true">نشط</SelectItem>
   421→   421→                <SelectItem value="false">مؤرشف</SelectItem>
   422→   422→              </SelectContent>
   423→   423→            </Select>
   424→   424→          </div>
   425→   425→        </CardContent>
   426→   426→      </Card>
   427→   427→
   428→   428→      {/* Loading / Error / Empty / Table */}
   429→   429→      {loading ? (
   430→   430→        <Card>
   431→   431→          <CardContent className="p-4 space-y-2">
   432→   432→            {[1, 2, 3, 4, 5].map((i) => <Skeleton key={i} className="h-12 w-full" />)}

   435→   435→      ) : error ? (
   436→   436→        <Card>
   437→   437→          <CardContent className="p-8 text-center">
   438→   438→            <AlertTriangle className="w-10 h-10 text-red-500 mx-auto mb-2" />

   449→   449→              <Plus className="w-4 h-4" />

   451→   451→            </Button>
   452→   452→          </CardContent>
   453→   453→        </Card>

   455→   455→        <Card>
